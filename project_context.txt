```python
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.database.database import get_db
from app.core.logger import logger
from app.schemas.health import HealthResponse

# Инициализация роутера с системным тегом для группировки в документации OpenAPI
router = APIRouter(prefix="", tags=["System"])


@router.get(
    "/health",
    response_model=HealthResponse,
    summary="Проверка статуса сервиса и подключения к БД",
)
async def health_check(
    session: AsyncSession = Depends(get_db),
) -> HealthResponse:
    """
    Диагностический эндпоинт для проверки статуса приложения и валидации соединения с PostgreSQL.
    """
    try:
        # Выполнение легковесного тестового запроса к PostgreSQL для проверки пула соединений
        await session.execute(text("SELECT 1"))

        # Возврат типизированного объекта HealthResponse в строгом соответствии со схемой
        return HealthResponse(
            status="ok",
            db_connected=True,
            version="1.0.0",
        )
    except Exception as error:
        # Фиксация полного трейса ошибки соединения в подсистеме централизованного логирования
        logger.error(
            "Сбой соединения с PostgreSQL: %s",
            error,
            exc_info=True,
        )

        # Выброс HTTP-исключения со статусом 500 для информирования вызывающей стороны
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Ошибка подключения к базе данных: {str(error)}",
        )
```

```python
from pydantic import Field
from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    """
    Конфигурация приложения и параметры окружения.
    Считывает значения из переменных среды или .env файла с валидацией через Pydantic.
    """

    # Системные метаданные сервиса с значениями по умолчанию
    app_title: str = Field(
        default="History Timeline API",
        description="Название сервиса",
    )
    app_version: str = Field(
        default="1.0.0",
        description="Текущая версия API",
    )
    debug: bool = Field(
        default=False,
        description="Режим отладки и расширенного логирования",
    )

    # Обязательные параметры подключения к PostgreSQL
    db_host: str = Field(
        ...,
        description="Хост базы данных PostgreSQL",
    )
    db_port: int = Field(
        ...,
        description="Порт базы данных PostgreSQL",
    )
    db_user: str = Field(
        ...,
        description="Имя пользователя PostgreSQL",
    )
    db_pass: str = Field(
        ...,
        description="Пароль пользователя PostgreSQL",
    )
    db_name: str = Field(
        ...,
        description="Имя базы данных PostgreSQL",
    )

    @property
    def database_url(self) -> str:
        """
        Формирование асинхронной строки подключения к СУБД для драйвера asyncpg.
        """
        return (
            f"postgresql+asyncpg://{self.db_user}:{self.db_pass}"
            f"@{self.db_host}:{self.db_port}/{self.db_name}"
        )

    # Конфигурация парсинга переменных окружения
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        case_sensitive=False,
        extra="ignore",
    )


# Инициализация глобального синглтона настроек
settings = Settings()
```

```python
import sys
from typing import Final

# Системное константное объявление поддерживаемой ветки среды исполнения
TARGET_PYTHON_BRANCH: Final[str] = "3.14"


def verify_python_environment() -> None:
    """
    Проверка соответствия версии интерпретатора установленным системным требованиям.
    Вызывается при старте FastAPI-приложения в рамках менеджера жизненного цикла lifespan.
    """
    # Считывание параметров мажорной и минорной версии CPython
    current_major: int = sys.version_info.major
    current_minor: int = sys.version_info.minor

    # Валидация запуска на целевой ветке Python 3.14 или выше
    if (current_major, current_minor) < (3, 14):
        raise RuntimeError(
            f"Несовместимая версия Python: {sys.version}. "
            f"Для работы сервиса требуется Python {TARGET_PYTHON_BRANCH} или выше."
        )


if __name__ == "__main__":
    # Локальный запуск проверки окружения из командной строки
    verify_python_environment()
    print(f"Среда успешно валидирована для ветки Python {TARGET_PYTHON_BRANCH}+.")
```

```python
from collections.abc import AsyncGenerator

from sqlalchemy.ext.asyncio import (
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)

from app.core.config.config import settings

# Создание асинхронного движка для подключения к PostgreSQL через драйвер asyncpg
engine = create_async_engine(
    url=settings.database_url,
    echo=False,
)

# Фабрика асинхронных сессий с отключенным сбросом состояния объектов после фиксации транзакции
async_session_maker = async_sessionmaker(
    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,
)


async def get_db() -> AsyncGenerator[AsyncSession, None]:
    """
    Асинхронный генератор зависимостей FastAPI для внедрения сессии базы данных.
    Предоставляет изолированную сессию, выполняет откат при исключении и гарантированно закрывает соединение.
    """
    session: AsyncSession = async_session_maker()
    try:
        # Передача сессии вызывающему эндпоинту или сервисному слою
        yield session
    except Exception:
        # Автоматический откат незавершенной транзакции при возникновении сбоя
        await session.rollback()
        # Повторный выброс исключения для дальнейшей обработки на уровне FastAPI
        raise
    finally:
        # Гарантированное закрытие сессии и возврат соединения в пул asyncpg
        await session.close()
```

```python
import logging
import sys

from app.core.config.config import settings

# Создание именованного логгера для логирования событий бизнес-логики сервиса
logger: logging.Logger = logging.getLogger("timeline_app")


def setup_logging() -> None:
    """
    Централизованная инициализация конфигурации логирования приложения.
    Настраивает корневой логгер, поток вывода sys.stdout, форматирование и фильтрацию шумных библиотек.
    """
    # Выбор базового уровня логирования в зависимости от режима отладки
    log_level = logging.DEBUG if settings.debug else logging.INFO

    # Шаблон форматирования: дата/время | уровень | модуль:строка - сообщение
    log_format = "%(asctime)s | %(levelname)-8s | %(name)s:%(lineno)d - %(message)s"
    date_format = "%Y-%m-%d %H:%M:%S"

    # Применение базовой конфигурации с принудительным переопределением обработчиков (force=True)
    logging.basicConfig(
        level=log_level,
        format=log_format,
        datefmt=date_format,
        stream=sys.stdout,
        force=True,
    )

    # Подавление избыточных информационных сообщений доступа сервера Uvicorn
    logging.getLogger("uvicorn.access").setLevel(logging.WARNING)

    # Отключение подробного SQL-трейсинга SQLAlchemy Engine в боевом режиме
    if not settings.debug:
        logging.getLogger("sqlalchemy.engine").setLevel(logging.WARNING)
```

```python
from collections.abc import AsyncGenerator
from contextlib import asynccontextmanager

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

from app.api.router import api_router
from app.core.config.config import settings
from app.core.config.python_spec import verify_python_environment
from app.core.logger import logger, setup_logging


@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncGenerator[None, None]:
    # Первичная инициализация подсистемы централизованного логирования
    setup_logging()

    # Валидация версии Python 3.14.7 перед выполнением основного рантайма
    verify_python_environment()

    # Фиксация события старта сервиса с параметрами конфигурации
    logger.info(
        "Запуск сервиса %s v%s на Python 3.14...",
        settings.app_title,
        settings.app_version,
    )

    # Передача управления работающему экземпляру приложения FastAPI
    yield

    # Фиксация события корректной остановки сервиса при завершении процесса
    logger.info("Сервис %s успешно остановлен.", settings.app_title)


# Инициализация основного приложения FastAPI с подключением менеджера жизненного цикла
app = FastAPI(
    title=settings.app_title,
    version=settings.app_version,
    lifespan=lifespan,
)

# Конфигурация CORS для разрешения кросс-доменных запросов на этапе локальной разработки
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Подключение центрального маршрутизатора API с версионированием v1
app.include_router(
    api_router,
    prefix="/api/v1",
)
```

```python
import re
from sqlalchemy.orm import DeclarativeBase, declared_attr


class Base(DeclarativeBase):
    """
    Базовый декларативный класс для всех ORM-моделей проекта.
    Автоматически формирует имя таблицы на основе имени Python-класса.
    """

    @declared_attr.directive
    def __tablename__(cls) -> str:
        # Преобразование имени класса из CamelCase в snake_case с помощью регулярного выражения
        snake_case_name = re.sub(r"(?<!^)(?=[A-Z])", "_", cls.__name__).lower()
        # Добавление суффикса 's' для формирования названия таблицы во множественном числе
        return f"{snake_case_name}s"
```

```python
from sqlalchemy import BIGINT, Integer, String, Text
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import Mapped, mapped_column

from app.models.base import Base


class Event(Base):
    """
    ORM-модель сущности исторического события.
    Хранит атрибуты временной шкалы, текстовые описания, категории и медиа-данные в формате JSONB.
    """

    # Уникальный первичный ключ события с индексом
    id: Mapped[int] = mapped_column(
        Integer,
        primary_key=True,
    )

    # Заголовок исторического события
    title: Mapped[str] = mapped_column(
        String(255),
        nullable=False,
    )

    # Развернутое описание исторического контекста (опционально)
    description: Mapped[str | None] = mapped_column(
        Text,
        nullable=True,
    )

    # Временная метка начала события (поддерживает отрицательные числа для эпохи до н.э.)
    start_date: Mapped[int] = mapped_column(
        BIGINT,
        nullable=False,
        index=True,
    )

    # Временная метка окончания события (для событий с длительной протяженностью)
    end_date: Mapped[int | None] = mapped_column(
        BIGINT,
        nullable=True,
        index=True,
    )

    # Категория события (например: политика, наука, культура)
    category: Mapped[str] = mapped_column(
        String(50),
        nullable=False,
```

```text
Как я это вижу: при создании карты задается начальная точка 1900г. После создания загружается экран с таймлайном, при этом отрезок от старта до текущего времени выделен например черным цветом, а отрезок (или луч) с текущего времени до, например 01.01.2100 светло серым, период до стариа можно обозначить таким же цветом. Ввод данных на таймлайн доступен в любой точке настоящего, прошлого или будущего. По ходу времени серый цвет будущего будет становиться черным настоящим.
По умолчанию создается таймлайн истории России 01.01.1900 по настоящее время.
Таким образом мы получаем временную шкалу, наполненную хронологически выстроенными событиями (От общих описаний, до детально расписанных поминутно).

Визуал:
События ранжируются по значимости - век, десятилетие, год, месяц, день, час, минута, секунда. 
Появляются на экране в зависимости от масштаба так же - век, десятилетие, год и т.д. 
Для того, чтобы обьединить множество подробностей одного события задается тэг по верхнему рангу события, например событие "Полет Гагарина" - вековое, поминутное представление полета с подзаголовками выглядит как (век)полет гагарина/(час)подъем, или (век)полет гагарина/(минута)"поехали!"
В любом приближении можно открыть дополнительное окно с представлением таймлайна в виде статьи (markdown?), где будут отражены события соответствующего ранга

Структура
1. Концепция и бизнес-цели веб-проекта
а) Назначение и ценность продукта
— Проект представляет собой интерактивную краудсорсинговую платформу для визуализации исторических процессов с глубокой хронологической детализацией.
— Главная ценность заключается в возможности непрерывного масштабирования временной шкалы от тысячелетий до секунд с точной привязкой текстового и мультимедийного контента. Возможность фиксации как масштабных исторических эпох, так и точечных краткосрочных событий в рамках единого интерфейса. 
— Система решает проблему фрагментированного восприятия истории, позволяя сопоставлять культурные, экономические и геополитические события на единой оси в наглядном виде.
б) Функциональные границы и ограничения
— В ядро системы (MVP) входит механика зуммирования шкалы, многопользовательская система ролей, панель модерации и генерация лонгридов в формате PDF.
— На текущем этапе исключаются алгоритмы автоматического фактчекинга и сложная система репутации авторов, требующие избыточных вычислительных мощностей. Но в будущем это можно добавить.
— Ключевые системные риски включают снижение производительности браузера при рендеринге тысяч перекрывающихся событий и проблему расхождения юлианского и григорианского календарей в истории России.
— Необходимо изначально заложить математическую модель базы данных для корректной обработки дат до нашей эры (отрицательные значения времени) и событий с неточной или длительной продолжительностью.
в) Профиль целевой аудитории и поведенческие сценарии
— Исследователи, историки и преподаватели, формирующие хронологические цепочки событий для лекционных и научных материалов
— Учащиеся и студенты, использующие интерактивную ось для визуального освоения исторических периодов и подготовки к аттестациям
— Контент-менеджеры и редакторы платформы, осуществляющие модерацию, валидацию и наполнение оси верифицированными данными
— Пользователи мобильных устройств и десктопов, требующие бесшовного жестового управления (pinch-to-zoom) для изменения масштаба времени
2. Пользовательские роли и сценарии взаимодействия
а) Модель ролей и прав доступа
— Гость платформы имеет права на просмотр шкалы, фильтрацию по категориям, навигацию по уровням детализации и экспорт PDF-документов.
— Зарегистрированный пользователь получает доступ к форме добавления новых событий с прикреплением медиафайлов, форматированием Markdown и указанием временных рамок.
— Администратор обладает правами модерации заявок, прямого добавления контента без очередей и настройки глобальных параметров (цветовые схемы, шрифты, расширение границ оси времени).
б) Детализация пользовательских путей
— Механика зуммирования (проваливания в века и секунды) требует реализации асинхронной подгрузки данных (lazy loading), чтобы запрашивать события исключительно для видимого отрезка оси, исключая перегрузку сети.
— При добавлении события пользователь выделяет отрезок на оси, выбирает категорию, вводит данные и обязательно указывает источник, после чего сущность переходит в статус ожидания.
— Для предотвращения тупиковых путей пользовательского опыта необходим постоянный элемент интерфейса в виде "хлебных крошек" или кнопки сброса масштаба, позволяющий мгновенно вернуться на уровень обзора столетия.
3. Информационная архитектура и структура данных
а) Модель данных и сущности
— Базовая сущность события включает идентификатор, точные метки начала и конца (timestamp), заголовок, описание, массив ссылок на медиа, категорию и статус модерации.
— Модель иерархических уровней вложенности оси Х: уровень 0 (столетия), уровень 1 (десятилетия), уровень 2 (годы), уровень 3 (месяцы), уровень 4 (дни), уровень 5 (часы/минуты/секунды).
— Логические связи типа один-ко-многим между временным отрезком верхнего уровня и вложенными подпериодами для корректного пересчета координат при зуме.
— Валидация требует проверки пересечения временных диапазонов, чтобы алгоритм компоновки мог автоматически разносить одновременные события по вертикали без визуальных наложений. Инструмент наложения (overlay) параллельно протекающих событий на оси Y (вертикальное стекирование блоков) во избежание визуального перекрытия элементов
— Хранение заявок реализуется через механизм теневых копий или версионирования, позволяющий администратору видеть разницу между опубликованной статьей и предложенными пользователем правками.
б) Экраны и компоновка интерфейса
— Главный экран представляет собой полноэкранный холст с элементами управления масштабом по краям и плавающими фильтрами категорий для снижения визуального шума.
— Панель модератора проектируется в виде таблицы с быстрыми фильтрами по статусам и встроенным окном предпросмотра, показывающим, как именно событие встанет на шкалу.
— Модуль выгрузки PDF работает через модальное окно выбора начального и конечного года, после чего алгоритм трансформирует горизонтальные данные в классический вертикальный лонгрид.
в) Функциональный состав и модульная структура
— Модуль динамического рендеринга горизонтальной оси Х с автоматическим пересчетом шага сетки разметки при изменении масштаба
— Система интерактивного погружения (drill-down) по клику на отрезок времени с плавной анимацией расширения текущего блока и скрытия соседних
— Функционал создания и редактирования исторических событий с привязкой к выбранному масштабу времени и указанием их точной протяженности на оси
4. Технический стек, интеграции и безопасность
а) Архитектурные и технологические требования
— Для рендеринга шкалы потребуется использование Canvas API, WebGL или специализированных библиотек визуализации (D3.js), так как стандартный DOM-дерево не справится с плавной анимацией при масштабировании.
— База данных должна поддерживать эффективное индексирование диапазонов (например, PostgreSQL с типами range), чтобы мгновенно отдавать события, попадающие в текущий фокус экрана.
— Процесс экспорта в PDF необходимо вынести на серверную сторону, чтобы избежать зависания клиентского устройства при формировании объемных лонгридов со встроенными изображениями.
б) Технологические параметры и внешние связи
— Интеграция со сторонними видеоплатформами через Embed API (YouTube, Вконтакте) для бесшовного воспроизведения медиаконтента внутри модальных окон
— Потоковая обработка и воспроизведение аудиофайлов (подкасты, лекции) с поддержкой стандартных форматов (MP3, AAC) через встроенный HTML5 Audio API
— Генератор экспортных файлов на стороне сервера (например, на базе библиотек Puppeteer или HTML2PDF) для динамической сборки и верстки PDF-документов в формате лонгрида
— Механизм динамического расширения границ временной шкалы в обе стороны (включая отрицательные значения Unix-времени для эпох до нашей эры)
в) Безопасность и соответствие регуляторным нормам
— Двухуровневая система аутентификации и авторизации пользователей через стандартные протоколы (JWT) или интеграцию с социальными сетями (VK ID)
— Модель разграничения прав доступа (RBAC): неавторизованный пользователь (просмотр), зарегистрированный пользователь (создание заявок), администратор (полный доступ)
— Обязательная фильтрация Markdown-разметки на стороне сервера перед записью в СУБД для предотвращения XSS-атак через инъекции вредоносных скриптов
— Система изоляции административной панели управления заявками, исключающая прямой доступ к эндпоинтам модерации со стороны обычных клиентов
5. Интерфейсная логика и визуальная структура
а) Навигационная карта и структура страниц
— Домашняя страница с центрированной интерактивной осью X, оснащенная контроллерами масштабирования и кнопками быстрого сброса к периоду по умолчанию (1900 — настоящее время)
— Модальный интерфейс или боковая панель (Drawer) для пошагового заполнения полей заявки на добавление события с выбором категории и загрузкой файлов
— Панель администратора со списком входящих заявок, разделенная по статусам (новые, на доработке, отклоненные) и функцией быстрого сравнения версий контента
— Экран настройки параметров отображения, позволяющий изменять цветовую схему категорий, шрифты интерфейса и типы иконок событий
б) Компонентная модель страниц
— Компонент временной сетки с поддержкой цветового кодирования направлений развития (культура, экономика, геополитика, наука) для визуальной сепарации данных
— Интерактивный блок события на таймлайне, меняющий свой размер и детализацию (от иконки до краткого превью) в зависимости от текущего шага зума
— Универсальный плеер-контейнер для воспроизведения прикрепленного аудио-, видео- и графического контента без перехода на внешние ресурсы
— Форма экспорта с чекбоксами выбора временных интервалов и категорий, генерирующая лонгрид-версию хроники для печати или сохранения
6. Стратегия поэтапной разработки (Дорожная карта MVP)
а) Границы минимально жизнеспособного продукта
— Реализация базового математического движка оси X с жестким масштабированием в диапазоне 1900 — 2026 гг. и вложенностью до уровня месяцев

— Спринт 1: создание строгой математической модели пересчета экранных координат во временные метки и реализация базового компонента бесконечной оси X с зумом. Разработка математической модели бесконечного таймлайна на фронтенде и структуры реляционной базы данных для хранения интервальных событий

в) Стратегия пошаговой сборки и тестирования
```

```text
Дорожная карта
1. Архитектурная база и серверное ядро
а) Подготовка фундамента
— Инициализация серверной части на FastAPI и настройка подключения к PostgreSQL.

— Проектирование схемы базы данных с учетом хранения дат в формате BIGINT для поддержки событий до нашей эры.

— Настройка базовых CI/CD пайплайнов для автоматического тестирования кода, сгенерированного ИИ.

б) Базовое API и маршрутизация
— Создание эндпоинтов для базовых CRUD-операций (создание, чтение, обновление, удаление) над историческими событиями.

— Реализация пространственно-временных SQL-запросов для выборки событий по диапазону дат.

— Подготовка пустого клиентского приложения с настроенным роутингом для последующего вайбкодинга интерфейса.

2. Математическая модель шкалы и базовый рендеринг (MVP 1.0)
а) Логика оси времени
— Разработка математического ядра на клиенте для пересчета экранных пикселей во временные метки.

— Реализация бесконечной горизонтальной оси X с динамическим изменением шага сетки (от столетий до секунд).

— Настройка жестового управления и зуммирования с помощью колеса мыши относительно положения курсора.

б) Отрисовка событий
— Внедрение алгоритма асинхронной подгрузки данных (lazy loading) при смещении или масштабировании шкалы.

— Вывод базовых маркеров событий на ось без сложного форматирования.

— Важным скрытым аспектом является управление памятью браузера, поэтому старые DOM-узлы или элементы Canvas должны удаляться при выходе за пределы экрана.

3. Компоновка интерфейса и предотвращение коллизий (MVP 2.0)
а) Вертикальное стекирование
— Создание алгоритма распределения событий по оси Y для параллельных или пересекающихся исторических периодов.

— Настройка автоматического расчета высоты блоков во избежание визуального наложения элементов друг на друга.

— Интеграция логики изменения плотности событий в зависимости от текущего уровня приближения (zoom level).

б) Детализация сущностей

— Подключение боковой панели или модального окна для просмотра полного описания события.

— Реализация иерархических тегов, позволяющих объединять мелкие события (минуты) под общим вековым событием.

4. Пользовательский контент и слои отображения
а) Управление направлениями
— Внедрение системы цветового кодирования категорий (культура, политика, экономика).

— Создание механизма изменения порядка слоев (z-index) и настройки их прозрачности.

— Добавление быстрых фильтров для скрытия нерелевантных направлений на клиенте.

б) Обогащение данных
— Интеграция Markdown-редактора для форматирования текста в описаниях событий.

— Настройка парсеров для корректного встраивания внешних медиафайлов (YouTube, VK) и аудиолекций.

5. Ролевая модель и система модерации
а) Аутентификация и безопасность
— Внедрение системы JWT-токенов и разделение прав на гостей, пользователей и администраторов.

— Реализация механизма теневых копий, при котором отредактированное или новое событие сохраняется в буферную таблицу.

б) Панель управления
— Разработка интерфейса модератора для просмотра, редактирования и одобрения заявок.

— Интеграция функционала мгновенной публикации событий администратором без прохождения очереди.

— Настройка параметров персонализации, где новый пользователь задает свою стартовую дату для шкалы.

6. Экспорт, локальность и децентрализация (Финальный релиз)
а) Генерация документов
— Создание серверного скрипта для конвертации горизонтальной выборки событий в вертикальный лонгрид.

— Настройка библиотеки для экспорта сформированного лонгрида в PDF-формат.

б) P2P и Local-first архитектура
— Внедрение технологии Service Workers и IndexedDB для кэширования приложения и обеспечения офлайн-работы.

— Подключение протоколов WebRTC для прямой синхронизации локальных баз данных между пользователями без участия центрального сервера.

— Этап децентрализации вынесен в конец, так как он кардинально меняет парадигму хранения данных и не должен тормозить релиз облачной версии.
```

# Маппинг расширений файлов к языкам разметки Markdown для подсветки синтаксиса
LANGUAGE_MAPPING: dict[str, str] = {
    ".py": "python",
    ".ini": "ini",
    ".txt": "text",
    ".md": "markdown",
    ".sql": "sql",
    ".yaml": "yaml",
    ".yml": "yaml",
    ".toml": "toml",
    ".json": "json",
    ".mako": "mako",
    ".html": "html",
    ".css": "css",
    ".js": "javascript",
}


def should_ignore_file(file_path: Path) -> bool:
    """Определяет, должен ли файл быть исключен из обработки."""
    if file_path.name in IGNORED_FILENAMES:
        return True
    if file_path.suffix.lower() in IGNORED_EXTENSIONS:
        return True
    return False


def get_syntax_language(file_path: Path) -> str:
    """Возвращает идентификатор языка для блока Markdown-подсветки."""
    if file_path.name == ".gitignore":
        return "gitignore"
    return LANGUAGE_MAPPING.get(file_path.suffix.lower(), "text")


def generate_ascii_tree(current_dir: Path, prefix: str = "") -> list[str]:
    """Рекурсивно формирует текстовое дерево каталогов проекта (ASCII Tree)."""
    tree_lines: list[str] = []
    try:
        entries = sorted(
            list(current_dir.iterdir()),
            key=lambda item: (item.is_file(), item.name.lower()),
        )
    except PermissionError:
        return tree_lines

    # Фильтрация объектов дерева
    valid_entries: list[Path] = []
    for entry in entries:
        if entry.is_dir() and entry.name in IGNORED_DIRS:
            continue
        if entry.is_file() and should_ignore_file(entry):
            continue
        valid_entries.append(entry)

    for index, entry in enumerate(valid_entries):
        is_last = index == len(valid_entries) - 1
        connector = "└── " if is_last else "├── "
        tree_lines.append(f"{prefix}{connector}{entry.name}")

        if entry.is_dir():
            nested_prefix = prefix + ("    " if is_last else "│   ")
            tree_lines.extend(generate_ascii_tree(entry, nested_prefix))

    return tree_lines


def scan_project_files(root_dir: Path) -> list[Path]:
    """Выполняет рекурсивный сбор всех подлежащих выгрузке файлов."""
    collected_files: list[Path] = []

    for dirpath, dirnames, filenames in os.walk(root_dir):
        # Модификация списка на месте для предотвращения обхода игнорируемых папок
        dirnames[:] = [d for d in dirnames if d not in IGNORED_DIRS]

        current_dir = Path(dirpath)
        for filename in sorted(filenames):
            file_path = current_dir / filename
            if not should_ignore_file(file_path):
                collected_files.append(file_path)

    return sorted(collected_files)


def main() -> None:
    """Точка входа: сборка контекста проекта в единый текстовый файл."""
    root_dir = Path(__file__).resolve().parent.parent
    output_file_path = root_dir / "project_context.txt"

    target_files = scan_project_files(root_dir)
    tree_structure = [f"{root_dir.name}/"] + generate_ascii_tree(root_dir)

    total_lines_count = 0

    with open(output_file_path, "w", encoding="utf-8", errors="replace") as report_file:
        # Формирование заголовка и схемы дерева файлов
        report_file.write("=" * 80 + "\n")
        report_file.write(f"СТРУКТУРА ПРОЕКТА: {root_dir.name}\n")
        report_file.write("=" * 80 + "\n\n")
        report_file.write("\n".join(tree_structure))
        report_file.write("\n\n\n")

        # Последовательная запись каждого файла проекта
        for file_path in target_files:
            relative_path = file_path.relative_to(root_dir).as_posix()
            file_extension = file_path.suffix if file_path.suffix else file_path.name
            syntax_lang = get_syntax_language(file_path)

            try:
                content = file_path.read_text(encoding="utf-8", errors="replace")
            except Exception as read_error:
                content = f"// Ошибка чтения содержимого файла: {str(read_error)}"

            file_lines = len(content.splitlines())
            total_lines_count += file_lines

            # Формирование метаданных и тела файла
            report_file.write("=" * 80 + "\n")
            report_file.write(f"Файл: {relative_path}\n")
            report_file.write(f"Формат: {file_extension}\n")
            report_file.write(f"Размер: {file_lines} строк\n")
            report_file.write("=" * 80 + "\n")
            report_file.write(f"```{syntax_lang}\n")
            report_file.write(content)
            if not content.endswith("\n"):
                report_file.write("\n")
            report_file.write("```\n\n")

    file_size_kb = output_file_path.stat().st_size / 1024

    # Вывод сводной технической информации в терминал
    print("\n" + "=" * 60)
    print("ВЫГРУЗКА КОНТЕКСТА ПРОЕКТА УСПЕШНО ЗАВЕРШЕНА")
    print("=" * 60)
    print(f"Обработано файлов     : {len(target_files)}")
    print(f"Всего строк кода      : {total_lines_count}")
    print(f"Размер выходного файла: {file_size_kb:.2f} KB")
    print(f"Расположение файла    : {output_file_path.resolve()}")
    print("=" * 60 + "\n")


if __name__ == "__main__":
    main()
